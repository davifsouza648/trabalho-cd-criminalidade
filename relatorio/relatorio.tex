\section{Dados e metodologia}\label{sec:metodo}
% ============================================================================

\subsection{Fontes dos dados}
\orientacao{Declarar as \textbf{duas} fontes de forma completa
(organizacao, link, data de download), como exige o enunciado:
\begin{itemize}
  \item \textbf{SSP-RS} -- Secretaria da Seguranca Publica do Rio Grande do Sul,
  microdados de ocorrencias (\url{https://ssp.rs.gov.br}), anos 2022--2025, todo
  o estado, granularidade de ocorrencia individual. Colunas usadas: data, hora,
  grupo, tipo de enquadramento, municipio, local e vitimas.
  \item \textbf{IBGE} -- populacao residente por municipio (Censo 2022) e a
  correspondencia municipio\,$\rightarrow$\,mesorregiao, ambas via API de dados
  do IBGE (Localidades e Agregados). Usadas para calcular a taxa por 100 mil
  habitantes e agregar por regiao.
\end{itemize}
Registrar a data de download de cada fonte.}

\subsection{Limpeza e preparacao}
\orientacao{Resumir o pipeline (\texttt{src/carga.py}, \texttt{src/ibge.py},
\texttt{src/limpeza.py}): (1) remocao de duplicatas por (Sequencia,
Enquadramento, Data); (2) atributos temporais -- ano, mes, dia da semana, fim
de semana e turno (madrugada/manha/tarde/noite); (3) normalizacao do nome do
municipio e \emph{join} com a tabela do IBGE (mesorregiao e populacao), com um
pequeno dicionario de correcao para as poucas grafias divergentes (ex.:
Sant'Ana do Livramento); (4) agrupamento dos tipos de enquadramento em
macrocategorias (ROUBO\_RUA, FURTO, LESAO, DROGAS, etc.) e marcacao de
``crimes de rua'' (sensiveis a patrulhamento ostensivo). Citar quantos
registros entraram/retidos e a cobertura do \emph{join} (praticamente 100\%).}

\subsection{Taxa por habitante}
\orientacao{Explicar por que a contagem bruta nao e comparavel entre municipios
(o maior sempre lidera) e como calculamos a \textbf{taxa media anual por 100 mil
habitantes}: ocorrencias somadas no periodo, divididas pela populacao (Censo
2022) e pelo numero de anos do recorte, vezes 100\,000. Assim o numerador
(varios anos) fica na mesma escala do denominador (um ano).}

\subsection{Metodo de analise}
\orientacao{Explicar e \textbf{justificar} as tecnicas (ponto que mais pesa na
nota):
\begin{itemize}
  \item \textbf{EDA}: estatistica descritiva + visualizacoes que respondem
  perguntas (Secao~\ref{sec:resultados}).
  \item \textbf{Qui-quadrado de independencia}: como ``tipo de ocorrencia'' e
  ``turno''/``mesorregiao'' sao variaveis \emph{categoricas}, o $\chi^2$ testa
  formalmente se ha associacao. $H_0$: independencia. Reportar $\chi^2$, graus
  de liberdade, $p$-valor, $V$ de Cramer e residuos padronizados.
  \item \textbf{Kruskal-Wallis}: teste nao-parametrico para verificar se a taxa
  por 100 mil hab. dos municipios difere entre as mesorregioes (nao assume
  normalidade).
  \item \textbf{(Opcional) KMeans dos municipios} por perfil de crime, para uma
  segmentacao de regioes com padrao semelhante.
\end{itemize}
Deixar explicito por que cada tecnica e coerente com a pergunta.}

\section{Limitacoes}\label{sec:limitacoes}
% ============================================================================
\orientacao{Discutir com honestidade (pesa na nota ``Excelente''):
\begin{itemize}
  \item \textbf{Ocorrencia registrada $\neq$ crime real}: a subnotificacao varia
  por tipo de crime e por municipio; o dado mede o que chega a policia.
  \item \textbf{Populacao de referencia}: a taxa usa a populacao residente do
  Censo 2022, nao a populacao \emph{presente}. Municipios litoraneos de veraneio
  aparecem com taxa inflada pela populacao flutuante do verao.
  \item \textbf{Municipios pequenos}: com poucos habitantes, poucos casos geram
  taxas instaveis (alta variancia).
  \item \textbf{Recorte de tempo e lugar}: conclusoes valem para o RS em
  2022--2025; nao extrapolam automaticamente para outros estados/periodos.
\end{itemize}}
